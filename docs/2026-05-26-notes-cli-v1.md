    id INTEGER PRIMARY KEY AUTOINCREMENT,
    subject TEXT NOT NULL,
    title TEXT NOT NULL DEFAULT '',
    started_at TEXT NOT NULL,
    status TEXT NOT NULL CHECK (status = 'active')

);

CREATE TABLE IF NOT EXISTS sessions (
    id INTEGER PRIMARY KEY AUTOINCREMENT,

    title TEXT NOT NULL DEFAULT '',

CREATE TABLE IF NOT EXISTS session_entries (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    session_id INTEGER NOT NULL,
    content TEXT NOT NULL,
    created_at TEXT NOT NULL,
    position INTEGER NOT NULL,
    FOREIGN KEY (session_id) REFERENCES sessions(id) ON DELETE CASCADE

package sqlite

import (
	"database/sql"
	"time"

	"github.com/miguel-warken/notes/internal/domain"
)

type DraftStore struct {
	db *sql.DB
}

func NewDraftStore(db *sql.DB) *DraftStore {
	return &DraftStore{db: db}
}

func (s *DraftStore) Create(subject, title string) (domain.DraftSession, error) {
	now := time.Now().UTC()

	result, err := s.db.Exec(
		`INSERT INTO draft_sessions(subject, title, started_at, status) VALUES (?, ?, ?, 'active')`,
		subject,
		title,
		now.Format(time.RFC3339Nano),
	)
	if err != nil {
		return domain.DraftSession{}, err
	}

	id, err := result.LastInsertId()
	if err != nil {
		return domain.DraftSession{}, err
	}

	return domain.DraftSession{
		ID:        id,
		Subject:   subject,
		Title:     title,
		StartedAt: now,
		Status:    "active",
	}, nil
}

func (s *DraftStore) AppendEntry(draftID int64, content string) error {
	_, err := s.db.Exec(
		`INSERT INTO draft_entries(draft_session_id, content, created_at) VALUES (?, ?, ?)`,
		draftID,
		content,
		time.Now().UTC().Format(time.RFC3339Nano),
	)
	return err
}

func (s *DraftStore) ListEntries(draftID int64) ([]domain.DraftEntry, error) {
	rows, err := s.db.Query(
		`SELECT id, draft_session_id, content, created_at FROM draft_entries WHERE draft_session_id = ? ORDER BY id`,
		draftID,
	)
	if err != nil {
		return nil, err
	}
	defer rows.Close()

	var entries []domain.DraftEntry
	for rows.Next() {
		var entry domain.DraftEntry
		var createdAt string

		if err := rows.Scan(&entry.ID, &entry.DraftSessionID, &entry.Content, &createdAt); err != nil {
			return nil, err
		}

		entry.CreatedAt, err = time.Parse(time.RFC3339Nano, createdAt)
		if err != nil {
			return nil, err
		}

		entries = append(entries, entry)
	}

	return entries, rows.Err()
}
```

func (s *DraftStore) GetActive() (domain.DraftSession, bool, error) {
	row := s.db.QueryRow(
		`SELECT id, subject, title, started_at, status FROM draft_sessions WHERE status = 'active' ORDER BY id DESC LIMIT 1`,
	)

	var draft domain.DraftSession
	var startedAt string
	err := row.Scan(&draft.ID, &draft.Subject, &draft.Title, &startedAt, &draft.Status)
	if err == sql.ErrNoRows {
		return domain.DraftSession{}, false, nil
	}
	if err != nil {
		return domain.DraftSession{}, false, err
	}

	draft.StartedAt, err = time.Parse(time.RFC3339Nano, startedAt)
	if err != nil {
		return domain.DraftSession{}, false, err
	}

	return draft, true, nil
}

func (s *DraftStore) DeleteDraft(draftID int64) error {
	_, err := s.db.Exec(`DELETE FROM draft_sessions WHERE id = ?`, draftID)
	return err
}

package app

import "github.com/miguel-warken/notes/internal/domain"

type sessionSaver interface {
	SaveDraft(draftID int64) (domain.Session, error)
}

type draftStore interface {
	Create(subject, title string) (domain.DraftSession, error)
	GetActive() (domain.DraftSession, bool, error)
	AppendEntry(draftID int64, content string) error
	DeleteDraft(draftID int64) error
}

type CaptureService struct {
	drafts draftStore
	saves  sessionSaver
}

func NewCaptureService(drafts draftStore, saves sessionSaver) *CaptureService {
	return &CaptureService{drafts: drafts, saves: saves}
}

func (s *CaptureService) Start(subject, title string) (domain.DraftSession, bool, error) {
	if draft, ok, err := s.drafts.GetActive(); err != nil {
		return domain.DraftSession{}, false, err
	} else if ok {
		return draft, true, nil
	}

	draft, err := s.drafts.Create(subject, title)
	return draft, false, err
}

func (s *CaptureService) Append(draftID int64, content string) error {
	return s.drafts.AppendEntry(draftID, content)
}

func (s *CaptureService) Cancel(draftID int64) error {
	return s.drafts.DeleteDraft(draftID)
}
```

import (
	"database/sql"
	"time"

	"github.com/miguel-warken/notes/internal/domain"

type SessionStore struct {
	db *sql.DB
}

func NewSessionStore(db *sql.DB) *SessionStore {
	return &SessionStore{db: db}
}

func (s *SessionStore) SaveDraft(draftID int64) (domain.Session, error) {
	tx, err := s.db.Begin()
	if err != nil {
		return domain.Session{}, err
	}
	defer tx.Rollback()

	var subject, title, startedAtRaw string
	err = tx.QueryRow(
		`SELECT subject, title, started_at FROM draft_sessions WHERE id = ?`,
		draftID,
	).Scan(&subject, &title, &startedAtRaw)
	if err != nil {
		return domain.Session{}, err
	}

	savedAt := time.Now().UTC()
	result, err := tx.Exec(
		`INSERT INTO sessions(subject, title, started_at, saved_at) VALUES (?, ?, ?, ?)`,
		subject,
		title,
		startedAtRaw,
		savedAt.Format(time.RFC3339Nano),
	)
	if err != nil {
		return domain.Session{}, err
	}

	sessionID, err := result.LastInsertId()
	if err != nil {
		return domain.Session{}, err
	}

	rows, err := tx.Query(
		`SELECT content, created_at FROM draft_entries WHERE draft_session_id = ? ORDER BY id`,
		draftID,
	)
	if err != nil {
		return domain.Session{}, err
	}
	defer rows.Close()

	position := 1
	for rows.Next() {
		var content, createdAt string
		if err := rows.Scan(&content, &createdAt); err != nil {
			return domain.Session{}, err
		}

		if _, err := tx.Exec(
			`INSERT INTO session_entries(session_id, content, created_at, position) VALUES (?, ?, ?, ?)`,
			sessionID,
			content,
			createdAt,
			position,
		); err != nil {
			return domain.Session{}, err
		}

		position++
	}
	if err := rows.Err(); err != nil {
		return domain.Session{}, err
	}

	if _, err := tx.Exec(`DELETE FROM draft_sessions WHERE id = ?`, draftID); err != nil {
		return domain.Session{}, err
	}

	if err := tx.Commit(); err != nil {
		return domain.Session{}, err
	}

	startedAt, err := time.Parse(time.RFC3339Nano, startedAtRaw)
	if err != nil {
		return domain.Session{}, err
	}

	return domain.Session{
		ID:        sessionID,
		Subject:   subject,
		Title:     title,
		StartedAt: startedAt,
		SavedAt:   savedAt,
	}, nil
}

func (s *SessionStore) ListEntries(sessionID int64) ([]domain.SessionEntry, error) {
	rows, err := s.db.Query(
		`SELECT id, session_id, content, created_at, position FROM session_entries WHERE session_id = ? ORDER BY position`,
		sessionID,
	)
	if err != nil {
		return nil, err
	}
	defer rows.Close()

	var entries []domain.SessionEntry
	for rows.Next() {
		var entry domain.SessionEntry
		var createdAt string

		if err := rows.Scan(&entry.ID, &entry.SessionID, &entry.Content, &createdAt, &entry.Position); err != nil {
			return nil, err
		}

		entry.CreatedAt, err = time.Parse(time.RFC3339Nano, createdAt)
		if err != nil {
			return nil, err
		}

		entries = append(entries, entry)
	}

	return entries, rows.Err()
}

package cli

import (
	"bufio"
	"fmt"
	"io"
	"strings"

	"github.com/miguel-warken/notes/internal/domain"
)

type captureRunner interface {
	Start(subject, title string) (draft domain.DraftSession, resumed bool, err error)
	Append(draftID int64, content string) error
	Cancel(draftID int64) error
	Save(draftID int64) (int64, error)
}

func runCapture(out io.Writer, prompts *Prompter, service captureRunner) error {
	subject, err := prompts.Ask("subject", true)
	if err != nil {
		return err
	}

	title, err := prompts.Ask("title", false)
	if err != nil {
		return err
	}

	draft, resumed, err := service.Start(subject, title)
	if err != nil {
		return err
	}

	if resumed {
		if _, err := fmt.Fprintln(out, "resumed existing draft"); err != nil {
			return err
		}
	}

	scanner := bufio.NewScanner(prompts.in)
	for scanner.Scan() {
		line := strings.TrimSpace(scanner.Text())

		switch line {
		case "":
			continue
		case "/cancel":
			return service.Cancel(draft.ID)
		case "/save":
			sessionID, err := service.Save(draft.ID)
			if err != nil {
				return err
			}
			_, err = fmt.Fprintf(out, "saved session %d\n", sessionID)
			return err
		default:
			if err := service.Append(draft.ID, line); err != nil {
				return err
			}
		}
	}

	if err := scanner.Err(); err != nil {
		return err
	}

	sessionID, err := service.Save(draft.ID)
	if err != nil {
		return err
	}

	_, err = fmt.Fprintf(out, "saved session %d\n", sessionID)
	return err
}
